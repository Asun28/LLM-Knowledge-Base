### AC12 — docstring audit script

**New file:** `scripts/audit_docstrings.py`.
**Logic per C-AC12-generator (R2-F16):**
```python
import ast, importlib, inspect, sys, re
import kb
offenders = []
for name in kb.__all__:
    try:
        obj = getattr(kb, name)
        src = inspect.getsource(obj)
        doc = inspect.getdoc(obj) or ""
        tree = ast.parse(src)
        fn = next((n for n in ast.walk(tree)
                   if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))), None)
        if fn is None: continue
        param_count = len([a for a in fn.args.args + fn.args.kwonlyargs])
        if param_count > 0 and not re.search(r"^\s*Args:", doc, re.M):
            offenders.append((name, "missing Args:"))
        if fn.returns and not (isinstance(fn.returns, ast.Constant) and fn.returns.value is None):
            if not re.search(r"^\s*Returns:", doc, re.M):
                offenders.append((name, "missing Returns:"))
        has_raise = any(isinstance(n, ast.Raise) for n in ast.walk(fn))
        if has_raise and not re.search(r"^\s*Raises:", doc, re.M):
            offenders.append((name, "missing Raises:"))
    except Exception as e:
        offenders.append((name, f"audit-error: {e}"))

if offenders:
    mode = "warning" if "--warn-only" in sys.argv else "error"
    print(f"::{mode}::{len(offenders)} docstring audit offenders")
    for name, reason in offenders:
        print(f"  {name}: {reason}")
    sys.exit(0 if mode == "warning" else 1)
```

**CI step:** warn-only or hard-fail per Task-0 decision.

### AC07 — duplicate-slug allowlist YAML externalization

**New file:** `src/kb/lint/_lint_yaml.py`:
```python
"""Lazy loader for ``wiki/_lint.yml`` operator-controlled lint config."""
from __future__ import annotations
import logging
from pathlib import Path
import yaml  # PyYAML

logger = logging.getLogger(__name__)

def load_lint_config(wiki_dir: Path) -> dict:
    """Read ``wiki_dir / "_lint.yml"`` if present; return parsed dict.

    Args:
        wiki_dir: Directory containing the optional `_lint.yml` config.

    Returns:
        dict with operator-supplied overrides; empty dict if file absent
        or unreadable. Always uses ``yaml.safe_load`` (never ``yaml.load``)
        per cycle-67 R2-F11 RCE-class threat.
    """
    path = wiki_dir / "_lint.yml"
    try:
        text = path.read_text(encoding="utf-8")
    except FileNotFoundError:
        return {}
    except (OSError, PermissionError) as exc:
        logger.warning("Failed to read %s: %s", path, exc)
        return {}
    try:
        loaded = yaml.safe_load(text)
    except yaml.YAMLError as exc:
        logger.warning("Failed to parse %s: %s", path, exc)
        return {}
    if not isinstance(loaded, dict):
        logger.warning("%s top-level is not a mapping; ignoring", path)
        return {}
    return loaded
```

**Production file:** `src/kb/utils/cli_backend.py` — replace `subprocess.run` body in `call_cli`:
```python
import sys, threading, time

def call_cli(prompt: str, *, backend: str, model: str, timeout: float) -> str:
    # ... existing prep + scrub ...
    proc = subprocess.Popen(
        cmd, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
        env=_scrub_env(backend),
    )
    out_buf = bytearray()
    err_buf = bytearray()
    overrun = threading.Event()

    def reader(stream, buf, cap):
        while True:
            chunk = stream.read(64 * 1024)
            if not chunk:
                return
            allowed = cap - len(buf)
            if allowed <= 0:
                overrun.set()
                return
            buf.extend(chunk[:allowed])
            if len(buf) >= cap:
                overrun.set()
                return

    out_thread = threading.Thread(target=reader, args=(proc.stdout, out_buf, MAX_CLI_STDOUT_BYTES), daemon=True)
    err_thread = threading.Thread(target=reader, args=(proc.stderr, err_buf, MAX_CLI_STDERR_BYTES), daemon=True)
    out_thread.start(); err_thread.start()

    if stdin_input is not None:
        try:
            proc.stdin.write(stdin_input)
        except (BrokenPipeError, OSError):
            pass
    proc.stdin.close()

    deadline = time.monotonic() + timeout
    while True:
        rc = proc.poll()
        if rc is not None:
            break
        if overrun.is_set():
            proc.terminate()
            if sys.platform.startswith("win"):
                try: proc.wait(timeout=0.5)
                except subprocess.TimeoutExpired: pass
            else:
                try: proc.wait(timeout=2)
                except subprocess.TimeoutExpired:
                    proc.kill()
                    try: proc.wait(timeout=0.5)
                    except subprocess.TimeoutExpired: pass
            break
        if time.monotonic() > deadline:
            proc.kill()
            raise LLMError(f"CLI timeout after {timeout}s for backend {backend!r}", kind="timeout")
        time.sleep(0.05)

    out_thread.join(timeout=1)
    err_thread.join(timeout=1)
    rc = proc.returncode
    stderr_text = bytes(err_buf).decode("utf-8", errors="replace")
    stderr_safe = _redact_secrets(stderr_text)
    if rc != 0:
        raise LLMError(
            f"CLI backend {backend!r} exited with code {rc}: {stderr_safe}"
        )
    if stderr_safe:
        logger.debug("cli_backend %s stderr: %s", backend, stderr_safe)
    raw_stdout = bytes(out_buf)
    stdout_text = raw_stdout.decode("utf-8", errors="replace")
    return _redact_secrets(_postprocess_stdout(backend, stdout_text)).strip()
```
